### 3.1 状态管理 (Zustand)

```typescript
// store/authStore.ts
import { create } from 'zustand';
import { persist } from 'zustand/middleware';

interface AuthState {
  teacher: Teacher | null;
  token: string | null;
  isAuthenticated: boolean;
  login: (email: string, password: string) => Promise<void>;
  logout: () => void;
}

export const useAuthStore = create<AuthState>()(
  persist(
    (set, get) => ({
      teacher: null,
      token: null,
      isAuthenticated: false,

      login: async (email, password) => {
        const response = await teacherApi.login({ email, password });
        set({
          teacher: response.teacher,
          token: response.token,
          isAuthenticated: true
        });
      },

      logout: () => {
        set({ teacher: null, token: null, isAuthenticated: false });
      }
    }),
    {
      name: 'auth-storage'
    }
  )
);

// store/escalationStore.ts
import { create } from 'zustand';

interface EscalationState {
  pendingEscalations: Escalation[];
  selectedEscalation: Escalation | null;
  setPendingEscalations: (escalations: Escalation[]) => void;
  addEscalation: (escalation: Escalation) => void;
  resolveEscalation: (id: string) => void;
  selectEscalation: (escalation: Escalation | null) => void;
}

export const useEscalationStore = create<EscalationState>((set) => ({
  pendingEscalations: [],
  selectedEscalation: null,

  setPendingEscalations: (escalations) =>
    set({ pendingEscalations: escalations }),

  addEscalation: (escalation) =>
    set((state) => ({
      pendingEscalations: [escalation, ...state.pendingEscalations]
    })),

  resolveEscalation: (id) =>
    set((state) => ({
      pendingEscalations: state.pendingEscalations.filter(e => e.id !== id)
    })),

  selectEscalation: (escalation) =>
    set({ selectedEscalation: escalation })
}));

// store/uiStore.ts - UI状态
interface UIState {
  sidebarCollapsed: boolean;
  notifications: Notification[];
  toggleSidebar: () => void;
  addNotification: (notification: Notification) => void;
  removeNotification: (id: string) => void;
}

export const useUIStore = create<UIState>((set) => ({
  sidebarCollapsed: false,
  notifications: [],

  toggleSidebar: () =>
    set((state) => ({ sidebarCollapsed: !state.sidebarCollapsed })),

  addNotification: (notification) =>
    set((state) => ({
      notifications: [notification, ...state.notifications]
    })),

  removeNotification: (id) =>
    set((state) => ({
      notifications: state.notifications.filter(n => n.id !== id)
    }))
}));
```

### 3.2 WebSocket Hook

```typescript
// hooks/useWebSocket.ts
import { useEffect, useRef, useCallback } from 'react';
import { io, Socket } from 'socket.io-client';
import { useAuthStore } from '../store/authStore';
import { useEscalationStore } from '../store/escalationStore';
import { useUIStore } from '../store/uiStore';

export const useWebSocket = () => {
  const socketRef = useRef<Socket | null>(null);
  const { teacher, token } = useAuthStore();
  const { addEscalation } = useEscalationStore();
  const { addNotification } = useUIStore();

  useEffect(() => {
    if (!teacher || !token) return;

    // 建立WebSocket连接
    socketRef.current = io('ws://localhost:8000', {
      query: { teacher_id: teacher.id },
      auth: { token }
    });

    const socket = socketRef.current;

    // 连接成功
    socket.on('connect', () => {
      console.log('WebSocket connected');
    });

    // 新的escalation
    socket.on('new_escalation', (data: Escalation) => {
      addEscalation(data);
      addNotification({
        id: data.id,
        type: 'escalation',
        title: '学生需要您的帮助',
        message: `${data.student_name} - ${data.context_summary}`,
        urgency: data.urgency,
        timestamp: new Date().toISOString()
      });
    });

    // 教材处理完成
    socket.on('material_processed', (data: { material_id: string; name: string }) => {
      addNotification({
        id: `mat_${data.material_id}`,
        type: 'success',
        title: '教材处理完成',
        message: `${data.name} 已可投入使用`,
        timestamp: new Date().toISOString()
      });
    });

    // 学生消息
    socket.on('student_message', (data: { student_id: string; content: string }) => {
      addNotification({
        id: `msg_${Date.now()}`,
        type: 'message',
        title: '收到学生消息',
        message: data.content.substring(0, 50) + '...',
        timestamp: new Date().toISOString()
      });
    });

    return () => {
      socket.disconnect();
    };
  }, [teacher, token, addEscalation, addNotification]);

  const sendMessage = useCallback((event: string, data: any) => {
    socketRef.current?.emit(event, data);
  }, []);

  return { socket: socketRef.current, sendMessage };
};
```

### 3.3 教材上传组件

```tsx
// components/materials/UploadModal.tsx
import React, { useState } from 'react';
import { Modal, Upload, Button, Form, Input, Select, Slider, Steps, message } from 'antd';
import { InboxOutlined, LoadingOutlined } from '@ant-design/icons';
import { useMutation, useQueryClient } from '@tanstack/react-query';
import { materialApi } from '../../services/materialApi';

const { Dragger } = Upload;
const { Step } = Steps;
const { Option } = Select;

interface UploadModalProps {
  visible: boolean;
  onCancel: () => void;
  classId: string;
}

export const UploadModal: React.FC<UploadModalProps> = ({
  visible,
  onCancel,
  classId
}) => {
  const [form] = Form.useForm();
  const [file, setFile] = useState<File | null>(null);
  const [currentStep, setCurrentStep] = useState(0);
  const [uploadId, setUploadId] = useState<string | null>(null);
  const queryClient = useQueryClient();

  // 上传mutation
  const uploadMutation = useMutation({
    mutationFn: materialApi.upload,
    onSuccess: (data) => {
      setUploadId(data.upload_id);
      setCurrentStep(1);
      message.success('上传成功，正在处理...');
      // 开始轮询处理进度
      pollProcessingStatus(data.upload_id);
    },
    onError: () => {
      message.error('上传失败');
    }
  });

  // 轮询处理进度
  const pollProcessingStatus = async (id: string) => {
    const checkStatus = async () => {
      const status = await materialApi.getUploadStatus(id);

      if (status.status === 'completed') {
        setCurrentStep(2);
        message.success('教材处理完成！');
        queryClient.invalidateQueries(['materials']);
      } else if (status.status === 'failed') {
        setCurrentStep(0);
        message.error('处理失败：' + status.error);
      } else {
        // 继续轮询
        setTimeout(checkStatus, 3000);
      }
    };

    checkStatus();
  };

  const handleUpload = () => {
    if (!file) {
      message.error('请先选择文件');
      return;
    }

    form.validateFields().then((values) => {
      const formData = new FormData();
      formData.append('file', file);
      formData.append('metadata', JSON.stringify({
        ...values,
        class_id: classId
      }));

      uploadMutation.mutate(formData);
    });
  };

  const steps = [
    {
      title: '选择文件',
      content: (
        <>
          <Dragger
            accept=".pdf,.docx,.ppt,.txt,.md"
            maxCount={1}
            beforeUpload={(file) => {
              setFile(file);
              return false; // 阻止自动上传
            }}
            onRemove={() => setFile(null)}
          >
            <p className="ant-upload-drag-icon">
              <InboxOutlined />
            </p>
            <p>点击或拖拽文件到此处</p>
            <p style={{ color: '#888' }}>
              支持 PDF, DOCX, PPT, TXT, Markdown
            </p>
          </Dragger>

          <Form
            form={form}
            layout="vertical"
            style={{ marginTop: 24 }}
          >
            <Form.Item
              name="material_name"
              label="教材名称"
              rules={[{ required: true }]}
            >
              <Input placeholder="例如：中三数学-因式分解" />
            </Form.Item>

            <Form.Item
              name="chapter_tags"
              label="章节标签"
            >
              <Select mode="tags" placeholder="添加标签">
                <Option value="因式分解">因式分解</Option>
                <Option value="十字相乘法">十字相乘法</Option>
              </Select>
            </Form.Item>

            <Form.Item
              name="difficulty"
              label="难度级别"
            >
              <Slider
                marks={{
                  0: '基础',
                  50: '中等',
                  100: '进阶'
                }}
              />
            </Form.Item>

            <Form.Item
              name="boundary_strictness"
              label="知识边界严格度"
              initialValue="moderate"
            >
              <Select>
                <Option value="strict">严格 - 只允许教材内内容</Option>
                <Option value="moderate">适中 - 教材为主，适度扩展</Option>
                <Option value="permissive">宽松 - 鼓励探索性学习</Option>
              </Select>
            </Form.Item>
          </Form>
        </>
      )
    },
    {
      title: '处理中',
      content: (
        <div style={{ textAlign: 'center', padding: 40 }}>
          <LoadingOutlined style={{ fontSize: 48 }} spin />
          <p style={{ marginTop: 16 }}>Pedagogical Architect 正在处理教材...</p>
          <p style={{ color: '#888' }}>解析内容、建立知识图谱、生成评估策略</p>
        </div>
      )
    },
    {
      title: '完成',
      content: (
        <div style={{ textAlign: 'center', padding: 40 }}>
          <h3>✅ 教材处理完成</h3>
          <p>知识图谱已生成，可以开始使用了</p>
          <Button
            type="primary"
            onClick={() => {
              onCancel();
              setCurrentStep(0);
              setFile(null);
            }}
          >
            查看教材
          </Button>
        </div>
      )
    }
  ];

  return (
    <Modal
      title="上传新教材"
      open={visible}
      onCancel={onCancel}
      width={700}
      footer={
        currentStep === 0 && (
          <Button
            type="primary"
            onClick={handleUpload}
            loading={uploadMutation.isPending}
          >
            开始上传
          </Button>
        )
      }
    >
      <Steps current={currentStep} style={{ marginBottom: 24 }}>
        {steps.map(step => (
          <Step key={step.title} title={step.title} />
        ))}
      </Steps>

      <div>{steps[currentStep].content}</div>
    </Modal>
  );
};
```

```tsx
import React, { useState } from 'react';
import { Modal, Radio, Input, Button, message, Descriptions, Timeline } from 'antd';
import { useMutation, useQueryClient } from '@tanstack/react-query';
import { escalationApi } from '../../services/escalationApi';
import { useEscalationStore } from '../../store/escalationStore';

const { TextArea } = Input;

export const ResponseModal: React.FC = () => {
  const { selectedEscalation, selectEscalation, resolveEscalation } = useEscalationStore();
  const [responseType, setResponseType] = useState<string>('guide_companion');
  const [message, setMessage] = useState('');
  const [strategyNote, setStrategyNote] = useState('');
  const queryClient = useQueryClient();

  const responseMutation = useMutation({
    mutationFn: escalationApi.respond,
    onSuccess: () => {
      message.success('响应已发送');
      resolveEscalation(selectedEscalation!.id);
      selectEscalation(null);
      queryClient.invalidateQueries(['escalations']);
    }
  });

    if (!selectedEscalation) return;

    const responseContent: any = {};

    if (responseType === 'take_over' || responseType === 'message_student') {

    if (responseType === 'guide_companion') {

      escalation_id: selectedEscalation.id,
      response_type: responseType,
      response_content: responseContent
    });
  };

  if (!selectedEscalation) return null;

  return (
    <Modal
      title="处理学生求助"
      open={!!selectedEscalation}
      onCancel={() => selectEscalation(null)}
      width={800}
      footer={[

          取消
        </Button>,
        <Button
          key="submit"
          type="primary"
          onClick={handleSubmit}
          loading={responseMutation.isPending}
        >
          确认响应
        </Button>
      ]}
    >
      <Descriptions bordered column={1} size="small" style={{ marginBottom: 24 }}>
        <Descriptions.Item label="学生">{selectedEscalation.student_name}</Descriptions.Item>
        <Descriptions.Item label="当前主题">{selectedEscalation.student_context?.current_topic}</Descriptions.Item>
        <Descriptions.Item label="错误次数">{selectedEscalation.student_context?.error_count}</Descriptions.Item>

// 工具函数
const formatTime = (isoString: string) => {
  const date = new Date(isoString);
  const now = new Date();
  const diff = Math.floor((now.getTime() - date.getTime()) / 1000 / 60);

  if (diff < 1) return '刚刚';
  if (diff < 60) return `${diff}分钟前`;
  if (diff < 1440) return `${Math.floor(diff / 60)}小时前`;
  return `${Math.floor(diff / 1440)}天前`;
};
```
